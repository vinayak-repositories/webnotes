        class Animal {}
        class Dog extends Animal {}

        let myDog = new Dog();

        console.log(myDog instanceof Dog);
        console.log(myDog instanceof Animal);
        console.log(myDog instanceof Object);

	d. delete -> To a delete a particular property from an object.
		     we can't delete the properties of built in objects because they are in [Read only Mode]

	syntax

	e. in 	-> It is a boolean operator that returns true if the given property  is available in specified object
			Note: even if the property value is undefined stil it returns true
		 
	syntax
		"property" in Object
