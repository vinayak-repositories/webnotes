setInterval()
 syntax: let intervalID = setInterval(function, delay, param1, param2, ...);
	function	   : The function to execute.
	delay		   : Time in milliseconds between each function call.
	param1, param2, ...: (Optional) Parameters to pass to the function.

- It would load the specified task into memory and release into process at regular intervals.
- It repeats the task untill removed from memory.
	


clearInterval() 
 syntax: clearInterval(intervalID);

- It is used to remove the task from memory using clearInterval()

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <style type="text/css">

    </style>
    <script type="text/javascript">

        function loadTime(){
            var now  = new Date();
            var time = now.toLocaleTimeString();
            console.log(time);
            document.querySelector("p").innerHTML = time;
        }

        var timer ;
        function bodyLoad() {
             timer = setInterval(loadTime,1000);
             
             document.getElementById("btnStop").addEventListener("click",function(){
                 console.log("stopped button clicked");
                 clearInterval(timer);
             })

             document.getElementById("btnStart").addEventListener("click",function(){
                console.log("start button clicked");
                timer = setInterval(loadTime,1000);
             })
        }
    </script>
</head>
<body onload="bodyLoad()">
    <div align="center" style="margin-top: 10px">
        <button id="btnStart">Start</button>
        <button id="btnStop">Stop</button>
    </div>
    <p align="center" style="color:red"></p>
</body>
</html>

            if (count==100) {

        }

        function loadClick(){
            //display progress bar

            document.getElementById("buttonContainer").style.display="none";

    </script>
