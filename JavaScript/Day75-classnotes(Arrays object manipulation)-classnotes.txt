Reading Array elements 
Adding elements to Array
Removing Array elements
	pop()     : It removes and returns the last element
	shift()   : It removes and returns the first element
        splice()  : It removes and returns the specified element

Syntax: 
 	arrayName.pop()
	arrayName.shift()
	arrayName.splice(startIndex,deleteCount)

Sorting Array elements
	sort()		: It arranges array elements in ascending order
	reverse()	: It arranges array elements in reverse order[bottom to top]

Syntax: 
	arrayName.sort()
	arrayName.reverse()

Searching for element in Array
	indexOf()	: It returns 
				a. first occurrence of indexnumber of element
				b. if not found return -1

	lastIndexOf()	: It returns the last occurrence of indexNumber of an element
	
	filter()	: It returns every occurance of element based on condition
	
	find()		: It returns first occurance of element based on condition

     <script type="text/javascript">
        var cities = ["Bengaluru","Hyderabad"];

        function loadCities() {
            document.getElementById("lstCities").innerHTML = "";
            document.getElementById("lblCount").innerHTML = `Total no of cities : ${cities.length}`

            cities.map(

                function(city){
                    console.log(city);

                    var option   = document.createElement("option")
                    option.value = city
                    option.text  = city
                    console.log(option)
                   
                    document.getElementById("lstCities").appendChild(option)
                }
          );
        }

        function addCities() {
            var cityName = document.getElementById("txtCity").value;
            if (cities.indexOf(cityName)==-1) {
                cities.push(cityName);
                loadCities();
                alert(`${cityName} added to list`)
                document.getElementById("txtCity").value=""         
            }else{
                alert(`${cityName} already exists`)
            }
        }

        function sortAsc() {
            cities.sort()
            loadCities()    
        }

        function sortDesc() {
            cities.sort()
            cities.reverse()
            loadCities()
        }

        function removeCity() {
            var cityName = document.getElementById("lstCities").value
            var index = cityName.indexOf()
            var flag = confirm(`Are you sure u want to delete ? ${cityName}`)
            if (flag) {
                cities.splice(index,1);
                loadCities()  
            } 
        }
        function removeAll() {

            var flag = confirm(`Are you sure u want to delete all cities ?`)
            if (flag) {
                cities.length = 0
                loadCities()  
            } 
        }
     </script>
